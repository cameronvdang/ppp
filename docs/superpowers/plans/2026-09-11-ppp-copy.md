- [ ] **Step 1: Write the failing guard test**

```ts
// app/src/copyGuard.test.ts
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join, resolve } from 'node:path'
import { describe, expect, it } from 'vitest'

const BANNED = [/IndexedDB/, /WebCrypto/, /non-extractable/, /\bAES\b/, /browser-managed/, /pseudonymous/, /subject ID/i, /external ID/i, /return URL/i, /client token/i, /single-owner/, /dedicated FinchNode/, /\bvault\b/i, /\bsealed\b/i, /service worker/i, /\bPWA\b/, /\bFHIR\b/, /\bRRULE\b/, /PRIVACY\.md/, /repository/i, /\bmetadata\b/i, /allowlist/i, /canonical/i, /signature verification/i, /\bBYOK\b/, /\bAPI\b/, /\.ics\b/]
const UI_DIRS = ['screens', 'components', 'privacy']

function* walk(dir: string): Generator<string> {
  for (const name of readdirSync(dir)) {
    const full = join(dir, name)
    if (statSync(full).isDirectory()) yield* walk(full)
    else if (/\.tsx?$/.test(name) && !/\.test\.tsx?$/.test(name)) yield full
  }
}

/** Only string literals and JSX text count; identifiers and comments do not. */
function userFacingText(source: string): string {
  const literals = [...source.matchAll(/'((?:[^'\\]|\\.)*)'|"((?:[^"\\]|\\.)*)"|`((?:[^`\\]|\\.)*)`/g)].map((m) => m[1] ?? m[2] ?? m[3] ?? '')
  const jsxText = [...source.matchAll(/>([^<>{}]+)</g)].map((m) => m[1])
  return [...literals, ...jsxText].filter((t) => /[A-Za-z]{3,}\s+[A-Za-z]{2,}/.test(t)).join('\n')
}

describe('UI copy guard', () => {
  it('keeps implementation vocabulary out of user-facing text', () => {
    const offenders: string[] = []
    for (const dir of UI_DIRS) for (const file of walk(resolve(__dirname, dir))) {
      const text = userFacingText(readFileSync(file, 'utf8'))
      for (const re of BANNED) for (const line of text.split('\n')) if (re.test(line)) offenders.push(`${file.split('/src/')[1]}: ${line.trim().slice(0, 90)}`)
    }
    expect([...new Set(offenders)]).toEqual([])
  })
})
```

Also add a sentence-length check in the same file: for the same text, split on `[.!?]` and fail on any sentence with more than 24 words (allows the consent sentence), reporting the file and sentence.
